export function distSq(p1: { x: number, y: number }, p2: { x: number, y: number }) {
  return (p1.x - p2.x) ** 2 + (p1.y - p2.y) ** 2;
}

export function dist(p1: { x: number, y: number }, p2: { x: number, y: number }) {
  return Math.sqrt(distSq(p1, p2));
}

// Distance from point to line segment
export function distToSegment(p: { x: number, y: number }, v: { x: number, y: number }, w: { x: number, y: number }) {
  const l2 = distSq(v, w);
  if (l2 === 0) return dist(p, v);
  let t = ((p.x - v.x) * (w.x - v.x) + (p.y - v.y) * (w.y - v.y)) / l2;
  t = Math.max(0, Math.min(1, t));
  return dist(p, { x: v.x + t * (w.x - v.x), y: v.y + t * (w.y - v.y) });
}

export function getArcParams(p0: { x: number, y: number }, p1: { x: number, y: number }, curve: number) {
  if (!curve || curve === 0) return null;
  const angle = (curve * Math.PI) / 180;
  const dx = p1.x - p0.x;
  const dy = p1.y - p0.y;
  const d = Math.hypot(dx, dy);
  if (d === 0) return null;

  const radius = Math.abs(d / (2 * Math.sin(angle / 2)));
  const mx = (p0.x + p1.x) / 2;
  const my = (p0.y + p1.y) / 2;
  const h = d / (2 * Math.tan(angle / 2));
  const nx = -dy / d;
  const ny = dx / d;

  const cx = mx + nx * h;
  const cy = my + ny * h;

  const startAngle = Math.atan2(p0.y - cy, p0.x - cx);
  const endAngle = Math.atan2(p1.y - cy, p1.x - cx);

  return { cx, cy, radius, startAngle, endAngle, counterClockwise: curve < 0 };
}

// Check if a point is close to an arc
export function distToArc(p: { x: number, y: number }, p0: { x: number, y: number }, p1: { x: number, y: number }, curve: number) {
  const arc = getArcParams(p0, p1, curve);
  if (!arc) return distToSegment(p, p0, p1);

  const distToCenter = dist(p, { x: arc.cx, y: arc.cy });
  const radialDist = Math.abs(distToCenter - arc.radius);
  
  // Quick angle check to see if point is within the arc sector
  let pAngle = Math.atan2(p.y - arc.cy, p.x - arc.cx);
  
  // Normalize angles to 0..2PI
  const normalize = (a: number) => (a < 0 ? a + 2 * Math.PI : a);
  let sa = normalize(arc.startAngle);
  let ea = normalize(arc.endAngle);
  let pa = normalize(pAngle);

  let within = false;
  if (arc.counterClockwise) {
    if (sa >= ea) within = (pa <= sa && pa >= ea);
    else within = (pa <= sa || pa >= ea);
  } else {
    if (ea >= sa) within = (pa >= sa && pa <= ea);
    else within = (pa >= sa || pa <= ea);
  }

  if (within) return radialDist;
  
  return Math.min(dist(p, p0), dist(p, p1));
}

type Action =

  | { type: 'UPDATE_STADIUM'; payload: Partial<Stadium>; pushHistory?: boolean }
  | { type: 'UPDATE_VERTEX'; index: number; payload: Partial<Stadium['vertexes'][0]> }
  | { type: 'UPDATE_SEGMENT'; index: number; payload: Partial<Stadium['segments'][0]> }
  | { type: 'UPDATE_DISC'; index: number; payload: Partial<Stadium['discs'][0]> }

  | { type: 'UPDATE_PLANE'; index: number; payload: Partial<Stadium['planes'][0]> }

  | { type: 'DELETE_SELECTION' }
  | { type: 'SET_BG_IMAGE'; payload: BgImageState | null }

  | { type: 'COPY_SELECTION' }

function reducer(state: EditorState, action: Action): EditorState {
  const pushHistory = (newState: EditorState, stadium: Stadium) => {
    const newHistory = state.history.slice(0, state.historyIndex + 1);
    newHistory.push(JSON.parse(JSON.stringify(stadium)));
    if (newHistory.length > 100) newHistory.shift();
    newState.history = newHistory;
    newState.historyIndex = newHistory.length - 1;
  };

  switch (action.type) {
    case 'SET_STADIUM': {
      const newState = { ...state, stadium: action.payload };
      if (action.pushHistory !== false) pushHistory(newState, action.payload);
      return newState;
    }
    case 'UPDATE_STADIUM': {
      const newStadium = { ...state.stadium, ...action.payload };
      const newState = { ...state, stadium: newStadium };
      if (action.pushHistory !== false) pushHistory(newState, newStadium);
      return newState;
    }
    case 'UPDATE_VERTEX': {
      const newVertexes = [...state.stadium.vertexes];
      newVertexes[action.index] = { ...newVertexes[action.index], ...action.payload };
      const newStadium = { ...state.stadium, vertexes: newVertexes };
      const newState = { ...state, stadium: newStadium };
      pushHistory(newState, newStadium);
      return newState;
    }
    case 'UPDATE_SEGMENT': {
      const newSegments = [...state.stadium.segments];
      newSegments[action.index] = { ...newSegments[action.index], ...action.payload };
      const newStadium = { ...state.stadium, segments: newSegments };
      const newState = { ...state, stadium: newStadium };
      pushHistory(newState, newStadium);
      return newState;
    }
    case 'UPDATE_DISC': {
      const newDiscs = [...state.stadium.discs];
      newDiscs[action.index] = { ...newDiscs[action.index], ...action.payload };
      const newStadium = { ...state.stadium, discs: newDiscs };
      const newState = { ...state, stadium: newStadium };
      pushHistory(newState, newStadium);
      return newState;
    }
    case 'UPDATE_GOAL': {
      const newGoals = [...state.stadium.goals];
      newGoals[action.index] = { ...newGoals[action.index], ...action.payload };
      const newStadium = { ...state.stadium, goals: newGoals };
      const newState = { ...state, stadium: newStadium };
      pushHistory(newState, newStadium);
      return newState;
    }
    case 'UPDATE_PLANE': {
      const newPlanes = [...state.stadium.planes];
      newPlanes[action.index] = { ...newPlanes[action.index], ...action.payload };
      const newStadium = { ...state.stadium, planes: newPlanes };
      const newState = { ...state, stadium: newStadium };
      pushHistory(newState, newStadium);
      return newState;
    }
    case 'DELETE_SELECTION': {
      if (state.selection.length === 0) return state;
      let newStadium = JSON.parse(JSON.stringify(state.stadium));

      // Sort descending to splice without shifting indices
      const toDelete = [...state.selection].sort((a, b) => b.index - a.index);

      // Delete segments first (no side effects on vertex indices at this point)
      for (const sel of toDelete) {
        if (sel.type === 'segment') newStadium.segments.splice(sel.index, 1);
      }

      // Delete discs and goals
      for (const sel of toDelete) {
        if (sel.type === 'disc') newStadium.discs.splice(sel.index, 1);
        if (sel.type === 'goal') newStadium.goals.splice(sel.index, 1);
        if (sel.type === 'plane') newStadium.planes.splice(sel.index, 1);
      }

      // Delete vertices and fix segment references
      for (const sel of toDelete) {
        if (sel.type === 'vertex') {
          newStadium.vertexes.splice(sel.index, 1);
          // Remove segments that referenced this vertex
          newStadium.segments = newStadium.segments.filter(
            (s: any) => s.v0 !== sel.index && s.v1 !== sel.index
          );
          // Decrement indices of remaining segments
          newStadium.segments.forEach((s: any) => {
            if (s.v0 > sel.index) s.v0--;
            if (s.v1 > sel.index) s.v1--;
          });
        }
      }

      const newState = { ...state, stadium: newStadium, selection: [] };
      pushHistory(newState, newStadium);
      return newState;
    }
    case 'UNDO': {
      if (state.historyIndex > 0) {
        return {
          ...state,
          historyIndex: state.historyIndex - 1,
          stadium: JSON.parse(JSON.stringify(state.history[state.historyIndex - 1])),
          selection: [],
        };
      }
      return state;
    }
    case 'REDO': {
      if (state.historyIndex < state.history.length - 1) {
        return {
          ...state,
          historyIndex: state.historyIndex + 1,
          stadium: JSON.parse(JSON.stringify(state.history[state.historyIndex + 1])),
          selection: [],
        };
      }
      return state;
    }
    case 'SET_SELECTION':
      return { ...state, selection: action.payload };
    case 'ADD_SELECTION':
      return { ...state, selection: [...state.selection, action.payload] };
    case 'SET_TOOL':
      return { ...state, tool: action.payload };
    case 'SET_THEME':
      return { ...state, theme: action.payload };
    case 'SET_CAMERA':
      return { ...state, ...action.payload };
    case 'SET_GRID':
      return { ...state, grid: action.payload };
    case 'SET_RULERS':
      return { ...state, showRulers: action.payload };
    case 'SET_SNAP':
      return { ...state, snapToGrid: action.payload.enabled, snapSize: action.payload.size };
    case 'SET_TEXT_MODE':
      return { ...state, textMode: action.payload };
    case 'SET_MOUSE':
      return { ...state, mouseX: action.payload.x, mouseY: action.payload.y };
    case 'SET_BG_IMAGE':
      return { ...state, bgImage: action.payload };
    case 'UPDATE_BG_IMAGE':
      return state.bgImage ? { ...state, bgImage: { ...state.bgImage, ...action.payload } } : state;

    case 'COPY_SELECTION': {
      if (state.selection.length === 0) return state;
      const items: ClipboardItem[] = [];
      for (const sel of state.selection) {
        if (sel.type === 'vertex') {
          items.push({ type: 'vertex', data: JSON.parse(JSON.stringify(state.stadium.vertexes[sel.index])) });
        }
        if (sel.type === 'disc') {
          items.push({ type: 'disc', data: JSON.parse(JSON.stringify(state.stadium.discs[sel.index])) });
        }
        if (sel.type === 'goal') {
          items.push({ type: 'goal', data: JSON.parse(JSON.stringify(state.stadium.goals[sel.index])) });
        }
        if (sel.type === 'segment') {
          const seg = state.stadium.segments[sel.index];
          if (!seg) continue;
          // Store full vertex position data so we can recreate them on paste
          const v0 = state.stadium.vertexes[seg.v0];
          const v1 = state.stadium.vertexes[seg.v1];
          items.push({
            type: 'segment',
            data: {
              ...JSON.parse(JSON.stringify(seg)),
              _v0: v0 ? JSON.parse(JSON.stringify(v0)) : null,
              _v1: v1 ? JSON.parse(JSON.stringify(v1)) : null,
            },
          });
        }
      }
      return { ...state, clipboard: items };
    }

    case 'PASTE_CLIPBOARD': {
      if (state.clipboard.length === 0) return state;
      const OFFSET = 20;
      const newStadium = JSON.parse(JSON.stringify(state.stadium));
      const newSelection: Selection[] = [];

      for (const item of state.clipboard) {
        if (item.type === 'vertex') {
          const idx = newStadium.vertexes.length;
          newStadium.vertexes.push({ ...item.data, x: (item.data.x ?? 0) + OFFSET, y: (item.data.y ?? 0) + OFFSET });
          newSelection.push({ type: 'vertex', index: idx });
        }
        if (item.type === 'disc') {
          const idx = newStadium.discs.length;
          const d = JSON.parse(JSON.stringify(item.data));
          if (d.pos) { d.pos[0] += OFFSET; d.pos[1] += OFFSET; }
          newStadium.discs.push(d);
          newSelection.push({ type: 'disc', index: idx });
        }
        if (item.type === 'goal') {
          const idx = newStadium.goals.length;
          const g = JSON.parse(JSON.stringify(item.data));
          if (g.p0) { g.p0[0] += OFFSET; g.p0[1] += OFFSET; }
          if (g.p1) { g.p1[0] += OFFSET; g.p1[1] += OFFSET; }
          newStadium.goals.push(g);
          newSelection.push({ type: 'goal', index: idx });
        }
        if (item.type === 'segment') {
          // Recreate both endpoint vertices, then the segment between them
          const v0idx = newStadium.vertexes.length;
          if (item.data._v0) {
            newStadium.vertexes.push({ ...item.data._v0, x: (item.data._v0.x ?? 0) + OFFSET, y: (item.data._v0.y ?? 0) + OFFSET });
          } else {
            newStadium.vertexes.push({ x: OFFSET, y: OFFSET });
          }
          const v1idx = newStadium.vertexes.length;
          if (item.data._v1) {
            newStadium.vertexes.push({ ...item.data._v1, x: (item.data._v1.x ?? 0) + OFFSET, y: (item.data._v1.y ?? 0) + OFFSET });
          } else {
            newStadium.vertexes.push({ x: OFFSET + 50, y: OFFSET });
          }
          const segIdx = newStadium.segments.length;
          const { _v0, _v1, ...segBase } = item.data;
          newStadium.segments.push({ ...segBase, v0: v0idx, v1: v1idx });
          newSelection.push({ type: 'segment', index: segIdx });
        }
      }

      const newState = { ...state, stadium: newStadium, selection: newSelection };
      pushHistory(newState, newStadium);
      return newState;
    }

    default:
      return state;
  }
}

const EditorContext = createContext<{
  state: EditorState;
  dispatch: React.Dispatch<Action>;
} | null>(null);

export const EditorProvider = ({ children }: { children: ReactNode }) => {
  const [state, dispatch] = useReducer(reducer, initialState);

  useEffect(() => {
    document.body.className = `theme-${state.theme === 'lightBlue' ? 'light-blue' : state.theme}`;
  }, [state.theme]);

  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) return;
      const ctrl = e.ctrlKey || e.metaKey;

      if (ctrl && e.key.toLowerCase() === 'z') {
        e.preventDefault();
        if (e.shiftKey) dispatch({ type: 'REDO' });
        else dispatch({ type: 'UNDO' });
        return;
      }
      if (ctrl && e.key.toLowerCase() === 'y') { e.preventDefault(); dispatch({ type: 'REDO' }); return; }
      if (ctrl && e.key.toLowerCase() === 'c') { e.preventDefault(); dispatch({ type: 'COPY_SELECTION' }); return; }
      if (ctrl && e.key.toLowerCase() === 'v') { e.preventDefault(); dispatch({ type: 'PASTE_CLIPBOARD' }); return; }

      switch (e.key.toLowerCase()) {
        case 's': dispatch({ type: 'SET_TOOL', payload: 'select' }); break;
        case 'h': dispatch({ type: 'SET_TOOL', payload: 'pan' }); break;
        case 'v': if (!ctrl) dispatch({ type: 'SET_TOOL', payload: 'vertex' }); break;
        case 'g': dispatch({ type: 'SET_TOOL', payload: 'segment' }); break;
        case 'd': if (!ctrl) dispatch({ type: 'SET_TOOL', payload: 'disc' }); break;
        case 'escape': dispatch({ type: 'SET_SELECTION', payload: [] }); break;
        case 'delete':
        case 'backspace':
          dispatch({ type: 'DELETE_SELECTION' });
          break;
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, []);

  return (
    <EditorContext.Provider value={{ state, dispatch }}>
      {children}
    </EditorContext.Provider>
  );
};

export const useEditor = () => {
  const context = useContext(EditorContext);
  if (!context) throw new Error('useEditor must be used within EditorProvider');
  return context;
};

/** Strip any internal editor fields (_v0, _v1, etc.) and produce clean .hbs JSON */
function buildCleanStadium(stadium: any) {
  return {
    ...stadium,
    vertexes: (stadium.vertexes ?? []).map((v: any) => {
      const { _v0, _v1, ...clean } = v;
      return clean;
    }),
    segments: (stadium.segments ?? []).map((s: any) => {
      const { _v0, _v1, ...clean } = s;
      return clean;
    }),
    discs: (stadium.discs ?? []).map((d: any) => {
      const { _v0, _v1, ...clean } = d;
      return clean;
    }),
  };
}

              <p className="text-[10px] font-bold text-muted-foreground uppercase tracking-wider">Object Counts</p>
              <CountBar label="Vertices" count={stadium.vertexes?.length ?? 0} />
              <CountBar label="Segments" count={stadium.segments?.length ?? 0} />

                <div><label className="text-[10px] text-muted-foreground mb-1 block">Height</label>
                  <input type="number" value={stadium.height} onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { height: Number(e.target.value) } })} className="w-full bg-background border px-2 py-1.5 rounded-lg text-sm" /></div>
              </div>
              <div><label className="text-[10px] text-muted-foreground mb-1 block">Spawn Distance</label>
                <input type="number" value={stadium.spawnDistance} onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { spawnDistance: Number(e.target.value) } })} className="w-full bg-background border px-2 py-1.5 rounded-lg text-sm" /></div>
            </div>

            <div className="space-y-2 border-t pt-3">

                className="w-full bg-background border px-3 py-2 rounded-lg text-sm focus:outline-none focus:border-primary"
              >
                <option value="grass">Grass</option>
                <option value="hockey">Hockey</option>
                <option value="none">None</option>
              </select>
              <div className="grid grid-cols-2 gap-2">
                <div><label className="text-[10px] text-muted-foreground mb-1 block">Width</label>
                  <input type="number" value={stadium.bg.width} onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { bg: { ...stadium.bg, width: Number(e.target.value) } } })} className="w-full bg-background border px-2 py-1.5 rounded text-sm" /></div>
                <div><label className="text-[10px] text-muted-foreground mb-1 block">Height</label>
                  <input type="number" value={stadium.bg.height} onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { bg: { ...stadium.bg, height: Number(e.target.value) } } })} className="w-full bg-background border px-2 py-1.5 rounded text-sm" /></div>
              </div>
              <div className="grid grid-cols-2 gap-2">
                <div><label className="text-[10px] text-muted-foreground mb-1 block">Kickoff R</label>
                  <input type="number" value={stadium.bg.kickOffRadius} onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { bg: { ...stadium.bg, kickOffRadius: Number(e.target.value) } } })} className="w-full bg-background border px-2 py-1.5 rounded text-sm" /></div>
                <div><label className="text-[10px] text-muted-foreground mb-1 block">Corner R</label>
                  <input type="number" value={stadium.bg.cornerRadius} onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { bg: { ...stadium.bg, cornerRadius: Number(e.target.value) } } })} className="w-full bg-background border px-2 py-1.5 rounded text-sm" /></div>
              </div>
              <div className="flex items-center gap-2">
                <input type="color" value={'#' + (stadium.bg.color || '718C5A')}
                  onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { bg: { ...stadium.bg, color: e.target.value.replace('#', '') } } })}
                  className="w-8 h-8 rounded border-none cursor-pointer" />

              </div>
            </div>

            <div className="space-y-2 border-t pt-3">
              <h3 className="text-[10px] font-bold text-muted-foreground uppercase tracking-wider flex items-center gap-1">
                <Settings2 className="w-3 h-3" /> Player Physics
              </h3>
              {Object.entries(stadium.playerPhysics).map(([key, val]) => (
                <div key={key} className="flex justify-between items-center gap-2">
                  <label className="text-xs text-muted-foreground truncate">{key}</label>
                  <input type="number" step="0.01" value={val as number}
                    onChange={e => dispatch({ type: 'UPDATE_STADIUM', payload: { playerPhysics: { ...stadium.playerPhysics, [key]: Number(e.target.value) } } })}
                    className="w-20 bg-background border px-2 py-1 rounded text-xs text-right" />
                </div>

        )}

        {/* ── IMAGE TAB ──────────────────────────────────── */}
        {tab === 'image' && (
          <>
            <div className="space-y-3">
              <h3 className="text-[10px] font-bold text-muted-foreground uppercase tracking-wider">Reference Image</h3>
              <p className="text-[10px] text-muted-foreground leading-relaxed">
                Load an image to trace over. Adjust opacity, scale, and position to align it with your map.
              </p>

              {/* Upload button */}
              <label className="flex items-center justify-center gap-2 w-full py-3 bg-primary/10 border border-primary/30 border-dashed rounded-xl cursor-pointer hover:bg-primary/20 transition-colors text-primary text-sm font-medium">
                <Upload className="w-4 h-4" />
                {bgImg ? 'Replace Image' : 'Choose Image'}
                <input type="file" accept="image/*" className="hidden" onChange={handleImageUpload} />
              </label>

              {bgImg && (
                <>
                  {/* Preview thumbnail */}
                  <div className="relative rounded-lg overflow-hidden border border-border bg-muted aspect-video flex items-center justify-center">
                    <img src={bgImg.url} alt="reference" className="max-h-full max-w-full object-contain" style={{ opacity: bgImg.opacity }} />
                  </div>

                  {/* Opacity */}
                  <div>
                    <div className="flex justify-between mb-1">
                      <label className="text-xs text-muted-foreground">Opacity</label>
                      <span className="text-xs font-mono text-foreground">{Math.round(bgImg.opacity * 100)}%</span>
                    </div>
                    <input type="range" min="0" max="1" step="0.01" value={bgImg.opacity}
                      onChange={e => dispatch({ type: 'UPDATE_BG_IMAGE', payload: { opacity: Number(e.target.value) } })} />
                  </div>

                  {/* Scale */}
                  <div>
                    <div className="flex justify-between mb-1">
                      <label className="text-xs text-muted-foreground">Scale</label>
                      <span className="text-xs font-mono text-foreground">{bgImg.scale.toFixed(2)}×</span>
                    </div>
                    <input type="range" min="0.05" max="5" step="0.01" value={bgImg.scale}
                      onChange={e => dispatch({ type: 'UPDATE_BG_IMAGE', payload: { scale: Number(e.target.value) } })} />
                    <input type="number" step="0.01" value={bgImg.scale}
                      onChange={e => dispatch({ type: 'UPDATE_BG_IMAGE', payload: { scale: Number(e.target.value) } })}
                      className="mt-1 w-full bg-background border px-2 py-1 rounded text-xs text-right" />
                  </div>

                  {/* Position */}
                  <div>
                    <label className="text-xs text-muted-foreground block mb-1">Position (world coords)</label>
                    <div className="grid grid-cols-2 gap-2">
                      <div>
                        <label className="text-[10px] text-muted-foreground">X</label>
                        <input type="number" value={Math.round(bgImg.x)}
                          onChange={e => dispatch({ type: 'UPDATE_BG_IMAGE', payload: { x: Number(e.target.value) } })}
                          className="w-full bg-background border px-2 py-1 rounded text-xs" />
                      </div>
                      <div>
                        <label className="text-[10px] text-muted-foreground">Y</label>
                        <input type="number" value={Math.round(bgImg.y)}
                          onChange={e => dispatch({ type: 'UPDATE_BG_IMAGE', payload: { y: Number(e.target.value) } })}
                          className="w-full bg-background border px-2 py-1 rounded text-xs" />
                      </div>
                    </div>
                  </div>

                  {/* Lock & Remove */}
                  <div className="flex gap-2">
                    <button
                      onClick={() => dispatch({ type: 'UPDATE_BG_IMAGE', payload: { locked: !bgImg.locked } })}
                      className={cn('flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg border text-xs font-medium transition-colors',
                        bgImg.locked ? 'bg-primary/20 border-primary text-primary' : 'bg-background border-border hover:bg-muted')}
                    >
                      {bgImg.locked ? <><Lock className="w-3.5 h-3.5" /> Locked</> : <><Unlock className="w-3.5 h-3.5" /> Unlocked</>}
                    </button>
                    <button
                      onClick={() => dispatch({ type: 'SET_BG_IMAGE', payload: null })}
                      className="flex items-center justify-center gap-1.5 px-3 py-2 rounded-lg border border-red-500/30 bg-red-500/10 text-red-400 hover:bg-red-500/20 text-xs font-medium transition-colors"
                    >
                      <Trash2 className="w-3.5 h-3.5" />
                    </button>
                  </div>

                  <p className="text-[10px] text-muted-foreground">
                    {bgImg.locked ? '🔒 Image is locked — drag moves the map, not the image.' : 'Unlock to drag the image with the pan tool.'}
                  </p>
                </>
              )}

              {!bgImg && (
                <div className="flex flex-col items-center justify-center py-8 text-center">
                  <ImageIcon className="w-10 h-10 text-muted-foreground/30 mb-3" />
                  <p className="text-xs text-muted-foreground">No reference image loaded.</p>
                  <p className="text-[10px] text-muted-foreground mt-1">Upload one to trace over it.</p>
                </div>
              )}
            </div>
          </>
        )}
      </div>
    </div>
  );
}

import React, { useRef, useEffect, useCallback, useState } from 'react';
import { useEditor, Selection } from '@/context/EditorContext';
import { distSq, distToSegment, distToArc, getArcParams } from '@/lib/math';
import { Copy, Trash2, Clipboard, MapPin } from 'lucide-react';

interface ContextMenuState {
  screenX: number;
  screenY: number;
  worldX: number;
  worldY: number;
  target: Selection | null;
}

export function CanvasRenderer() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const { state, dispatch } = useEditor();

  const isDraggingRef = useRef(false);
  const dragStartRef = useRef<{ x: number; y: number } | null>(null);
  const tempVertexRef = useRef<number | null>(null);
  const isPanningRef = useRef(false);
  const panStartRef = useRef<{ x: number; y: number; camX: number; camY: number } | null>(null);
  const rubberBandRef = useRef<{ sx: number; sy: number; ex: number; ey: number } | null>(null);
  const rubberBandActiveRef = useRef(false);
  const bgImgElementRef = useRef<HTMLImageElement | null>(null);
  const bgImgUrlRef = useRef<string>('');
  const stateRef = useRef(state);
  useEffect(() => { stateRef.current = state; }, [state]);

  const [contextMenu, setContextMenu] = useState<ContextMenuState | null>(null);
  const [posEdit, setPosEdit] = useState<{ x: string; y: string } | null>(null);

  // Preload bg image
  useEffect(() => {
    const bg = state.bgImage;
    if (!bg) { bgImgElementRef.current = null; bgImgUrlRef.current = ''; return; }
    if (bg.url === bgImgUrlRef.current) return;
    const img = new Image();
    img.onload = () => { bgImgElementRef.current = img; bgImgUrlRef.current = bg.url; };
    img.src = bg.url;
  }, [state.bgImage?.url]);

  // Resize
  useEffect(() => {
    const handleResize = () => {
      const c = canvasRef.current;
      if (!c) return;
      c.width = c.parentElement!.clientWidth;
      c.height = c.parentElement!.clientHeight;
    };
    handleResize();
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  }, []);

  // Main render loop
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;
    let animId: number;

    const render = () => {
      const s = stateRef.current;
      const { stadium, zoom, camX, camY, grid, theme, selection, showRulers } = s;
      const cw = canvas.width, ch = canvas.height;
      const bgColor = theme === 'dark' ? '#1a1b26' : theme === 'neon' ? '#050505' : '#f8fafc';
      const gridColor = theme === 'neon' ? '#222' : theme === 'dark' ? '#272a3b' : '#e2e8f0';
      const selColor = theme === 'neon' ? '#00ffff' : '#facc15';
      const rulerBg = theme === 'dark' ? '#12131d' : theme === 'neon' ? '#0a0a0a' : '#e2e8f0';
      const rulerText = theme === 'dark' ? '#6b7280' : theme === 'neon' ? '#444' : '#94a3b8';
      const RULER = showRulers ? 20 : 0;

      ctx.fillStyle = bgColor;
      ctx.fillRect(0, 0, cw, ch);

      // World-to-screen (accounting for ruler offset)
      const w2s = (wx: number, wy: number) => ({
        x: RULER + (cw - RULER) / 2 + (wx - camX) * zoom,
        y: RULER + (ch - RULER) / 2 + (wy - camY) * zoom,
      });

      // ── Grid ─────────────────────────────────────
      if (grid) {
        const step = 50 * zoom;
        const ox = (((cw - RULER) / 2 - camX * zoom) % step + step) % step;
        const oy = (((ch - RULER) / 2 - camY * zoom) % step + step) % step;
        ctx.strokeStyle = gridColor;
        ctx.lineWidth = 1;
        ctx.beginPath();
        for (let x = RULER + ox; x < cw; x += step) { ctx.moveTo(x, RULER); ctx.lineTo(x, ch); }
        for (let y = RULER + oy; y < ch; y += step) { ctx.moveTo(RULER, y); ctx.lineTo(cw, y); }
        ctx.stroke();
        // Origin cross
        const o = w2s(0, 0);
        ctx.strokeStyle = theme === 'neon' ? '#ff00ff' : '#ef4444';
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(o.x - 10, o.y); ctx.lineTo(o.x + 10, o.y);
        ctx.moveTo(o.x, o.y - 10); ctx.lineTo(o.x, o.y + 10);
        ctx.stroke();
      }

      // ── Stadium field ─────────────────────────────
      const bg = stadium.bg;
      if (bg && bg.type !== 'none') {
        const p1 = w2s(-bg.width, -bg.height);
        const p2 = w2s(bg.width, bg.height);
        ctx.fillStyle = `#${bg.color || '718C5A'}`;
        if (bg.cornerRadius > 0) {
          ctx.beginPath();
          ctx.roundRect(p1.x, p1.y, p2.x - p1.x, p2.y - p1.y, bg.cornerRadius * zoom);
          ctx.fill();
        } else {
          ctx.fillRect(p1.x, p1.y, p2.x - p1.x, p2.y - p1.y);
        }
        if (bg.kickOffRadius > 0) {
          const center = w2s(0, 0);
          ctx.strokeStyle = '#ffffff'; ctx.lineWidth = 2;
          ctx.beginPath(); ctx.arc(center.x, center.y, bg.kickOffRadius * zoom, 0, Math.PI * 2); ctx.stroke();
          ctx.beginPath(); ctx.moveTo(center.x, p1.y); ctx.lineTo(center.x, p2.y); ctx.stroke();
        }
      }

      // ── Reference image (ON TOP of field) ─────────
      const bgI = s.bgImage;
      if (bgI && bgImgElementRef.current) {
        const img = bgImgElementRef.current;
        const sc = bgI.scale * zoom;
        const sx = RULER + (cw - RULER) / 2 + (bgI.x - camX) * zoom;
        const sy = RULER + (ch - RULER) / 2 + (bgI.y - camY) * zoom;
        ctx.save();
        ctx.globalAlpha = bgI.opacity;
        ctx.drawImage(img, sx - (img.width * sc) / 2, sy - (img.height * sc) / 2, img.width * sc, img.height * sc);
        ctx.restore();
      }

      // ── Goals ─────────────────────────────────────
      stadium.goals?.forEach((goal, i) => {
        if (!goal.p0 || !goal.p1) return;
        const gp0 = w2s(goal.p0[0], goal.p0[1]);
        const gp1 = w2s(goal.p1[0], goal.p1[1]);
        const isSel = selection.some(s => s.type === 'goal' && s.index === i);
        ctx.beginPath(); ctx.moveTo(gp0.x, gp0.y); ctx.lineTo(gp1.x, gp1.y);
        ctx.strokeStyle = isSel ? selColor : (goal.team === 'red' ? '#ff4444' : '#4488ff');
        ctx.lineWidth = isSel ? 5 : 3; ctx.stroke();
      });

      // ── Segments ──────────────────────────────────
      stadium.segments?.forEach((seg, i) => {
        if (seg.v0 == null || seg.v1 == null) return;
        const v0 = stadium.vertexes[seg.v0], v1 = stadium.vertexes[seg.v1];
        if (!v0 || !v1) return;
        const p0 = w2s(v0.x, v0.y), p1 = w2s(v1.x, v1.y);
        const isSel = selection.some(s => s.type === 'segment' && s.index === i);
        ctx.beginPath();
        if (!seg.curve || seg.curve === 0) { ctx.moveTo(p0.x, p0.y); ctx.lineTo(p1.x, p1.y); }
        else {
          const arc = getArcParams(v0, v1, seg.curve);
          if (arc) { const ac = w2s(arc.cx, arc.cy); ctx.arc(ac.x, ac.y, arc.radius * zoom, arc.startAngle, arc.endAngle, arc.counterClockwise); }
          else { ctx.moveTo(p0.x, p0.y); ctx.lineTo(p1.x, p1.y); }
        }
        ctx.strokeStyle = isSel ? selColor : `#${seg.color || 'FFFFFF'}`;
        ctx.lineWidth = isSel ? 4 : 2;
        ctx.setLineDash(seg.vis === false ? [5, 5] : []);
        ctx.stroke(); ctx.setLineDash([]);
      });

      // ── Discs ─────────────────────────────────────
      stadium.discs?.forEach((disc, i) => {
        if (!disc.pos || disc.pos[0] == null) return;
        const p = w2s(disc.pos[0], disc.pos[1]);
        const isSel = selection.some(s => s.type === 'disc' && s.index === i);
        const r = Math.max((disc.radius ?? 7) * zoom, 1);
        ctx.beginPath(); ctx.arc(p.x, p.y, r, 0, Math.PI * 2);
        ctx.fillStyle = `#${disc.color || 'FFAA00'}`; ctx.fill();
        ctx.lineWidth = isSel ? 4 : 2;
        ctx.strokeStyle = isSel ? selColor : 'rgba(0,0,0,0.5)'; ctx.stroke();
      });

      // ── Vertexes ──────────────────────────────────
      stadium.vertexes?.forEach((v, i) => {
        if (v.x == null) return;
        const p = w2s(v.x, v.y);
        const isSel = selection.some(s => s.type === 'vertex' && s.index === i);
        ctx.beginPath(); ctx.arc(p.x, p.y, isSel ? 7 : 4, 0, Math.PI * 2);
        ctx.fillStyle = isSel ? selColor : '#ffffff'; ctx.fill();
        ctx.strokeStyle = '#000'; ctx.lineWidth = 1; ctx.stroke();
      });

      // ── Segment creation preview ───────────────────
      const tv = tempVertexRef.current;
      if (s.tool === 'segment' && tv !== null) {
        const v0 = stadium.vertexes[tv];
        if (v0) {
          const p0 = w2s(v0.x, v0.y), mPos = w2s(s.mouseX, s.mouseY);
          ctx.beginPath(); ctx.moveTo(p0.x, p0.y); ctx.lineTo(mPos.x, mPos.y);
          ctx.strokeStyle = selColor; ctx.lineWidth = 2;
          ctx.setLineDash([4, 4]); ctx.stroke(); ctx.setLineDash([]);
        }
      }

      // ── Rubber-band selection ──────────────────────
      const rb = rubberBandRef.current;
      if (rubberBandActiveRef.current && rb) {
        const rx = Math.min(rb.sx, rb.ex), ry = Math.min(rb.sy, rb.ey);
        const rw = Math.abs(rb.ex - rb.sx), rh = Math.abs(rb.ey - rb.sy);
        ctx.strokeStyle = selColor; ctx.lineWidth = 1;
        ctx.setLineDash([4, 2]); ctx.strokeRect(rx, ry, rw, rh);
        ctx.fillStyle = theme === 'neon' ? 'rgba(0,255,255,0.05)' : 'rgba(250,204,21,0.07)';
        ctx.fillRect(rx, ry, rw, rh); ctx.setLineDash([]);
      }

      // ── Coordinate Rulers ──────────────────────────
      if (showRulers && RULER > 0) {
        // Background strips
        ctx.fillStyle = rulerBg;
        ctx.fillRect(0, 0, cw, RULER); // top
        ctx.fillRect(0, 0, RULER, ch); // left
        ctx.fillRect(0, 0, RULER, RULER); // corner

        const step = pickRulerStep(zoom);
        const startWorldX = Math.floor((camX - (cw - RULER) / 2 / zoom) / step) * step;
        const startWorldY = Math.floor((camY - (ch - RULER) / 2 / zoom) / step) * step;

        ctx.font = '9px monospace';
        ctx.fillStyle = rulerText;
        ctx.textBaseline = 'middle';
        ctx.textAlign = 'center';

        // Top ruler (X axis)
        for (let wx = startWorldX; wx < camX + (cw - RULER) / 2 / zoom + step; wx += step) {
          const sx = RULER + (cw - RULER) / 2 + (wx - camX) * zoom;
          if (sx < RULER || sx > cw) continue;
          ctx.fillStyle = rulerText;
          ctx.fillRect(sx, RULER - 5, 1, 5);
          ctx.fillText(String(Math.round(wx)), sx, RULER / 2);
        }

        // Left ruler (Y axis)
        ctx.textAlign = 'right';
        ctx.textBaseline = 'middle';
        for (let wy = startWorldY; wy < camY + (ch - RULER) / 2 / zoom + step; wy += step) {
          const sy = RULER + (ch - RULER) / 2 + (wy - camY) * zoom;
          if (sy < RULER || sy > ch) continue;
          ctx.fillStyle = rulerText;
          ctx.fillRect(RULER - 5, sy, 5, 1);
          ctx.fillText(String(Math.round(wy)), RULER - 7, sy);
        }

        // Cursor position indicator on rulers
        const msx = RULER + (cw - RULER) / 2 + (s.mouseX - camX) * zoom;
        const msy = RULER + (ch - RULER) / 2 + (s.mouseY - camY) * zoom;
        ctx.fillStyle = selColor;
        ctx.fillRect(msx - 1, 0, 2, RULER);
        ctx.fillRect(0, msy - 1, RULER, 2);

        // Corner label
        ctx.fillStyle = rulerText;
        ctx.textAlign = 'center';
        ctx.textBaseline = 'middle';
        ctx.fillText('xy', RULER / 2, RULER / 2);

        // Divider lines
        ctx.strokeStyle = theme === 'dark' ? '#2a2c3e' : theme === 'neon' ? '#1a1a1a' : '#cbd5e1';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(RULER, 0); ctx.lineTo(RULER, ch);
        ctx.moveTo(0, RULER); ctx.lineTo(cw, RULER);
        ctx.stroke();
      }

      animId = requestAnimationFrame(render);
    };

    render();
    return () => cancelAnimationFrame(animId);
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  function pickRulerStep(zoom: number): number {
    const candidates = [1, 2, 5, 10, 20, 25, 50, 100, 200, 250, 500, 1000];
    const targetPx = 60;
    for (const c of candidates) {
      if (c * zoom >= targetPx) return c;
    }
    return 1000;
  }

  const screenToWorld = useCallback((ex: number, ey: number) => {
    const canvas = canvasRef.current!;
    const rect = canvas.getBoundingClientRect();
    const { zoom, camX, camY, showRulers } = stateRef.current;
    const RULER = showRulers ? 20 : 0;
    const sx = ex - rect.left - RULER;
    const sy = ey - rect.top - RULER;
    const cw = canvas.width - RULER, ch = canvas.height - RULER;
    let wx = sx / zoom + camX - cw / 2 / zoom;
    let wy = sy / zoom + camY - ch / 2 / zoom;
    // Snap to grid if enabled
    if (stateRef.current.snapToGrid) {
      const snap = stateRef.current.snapSize;
      wx = Math.round(wx / snap) * snap;
      wy = Math.round(wy / snap) * snap;
    }
    return { x: wx, y: wy };
  }, []);

  const screenCoordsRaw = useCallback((ex: number, ey: number) => {
    const canvas = canvasRef.current!;
    const rect = canvas.getBoundingClientRect();
    return { sx: ex - rect.left, sy: ey - rect.top };
  }, []);

  const getHoveredObject = useCallback((wx: number, wy: number) => {
    const { stadium, zoom } = stateRef.current;
    const thresh = 10 / zoom;
    for (let i = (stadium.vertexes?.length ?? 0) - 1; i >= 0; i--) {
      const v = stadium.vertexes[i];
      if (v && distSq({ x: wx, y: wy }, v) < thresh * thresh) return { type: 'vertex' as const, index: i };
    }
    for (let i = (stadium.discs?.length ?? 0) - 1; i >= 0; i--) {
      const d = stadium.discs[i];
      if (!d?.pos) continue;
      const r = (d.radius ?? 7) + thresh;
      if (distSq({ x: wx, y: wy }, { x: d.pos[0], y: d.pos[1] }) < r * r) return { type: 'disc' as const, index: i };
    }
    for (let i = (stadium.goals?.length ?? 0) - 1; i >= 0; i--) {
      const g = stadium.goals[i];
      if (!g?.p0 || !g?.p1) continue;
      if (distToSegment({ x: wx, y: wy }, { x: g.p0[0], y: g.p0[1] }, { x: g.p1[0], y: g.p1[1] }) < thresh)
        return { type: 'goal' as const, index: i };
    }
    for (let i = (stadium.segments?.length ?? 0) - 1; i >= 0; i--) {
      const seg = stadium.segments[i];
      if (seg.v0 == null || seg.v1 == null) continue;
      const v0 = stadium.vertexes[seg.v0], v1 = stadium.vertexes[seg.v1];
      if (!v0 || !v1) continue;
      const d = (!seg.curve || seg.curve === 0) ? distToSegment({ x: wx, y: wy }, v0, v1) : distToArc({ x: wx, y: wy }, v0, v1, seg.curve);
      if (d < thresh) return { type: 'segment' as const, index: i };
    }
    return null;
  }, []);

  const getObjectsInRect = useCallback((x1: number, y1: number, x2: number, y2: number) => {
    const { stadium } = stateRef.current;
    const minX = Math.min(x1, x2), maxX = Math.max(x1, x2);
    const minY = Math.min(y1, y2), maxY = Math.max(y1, y2);
    const inside = (wx: number, wy: number) => wx >= minX && wx <= maxX && wy >= minY && wy <= maxY;

    const result: Array<{ type: 'vertex' | 'segment' | 'disc' | 'goal'; index: number }> = [];
    const selectedSegmentVertexes = new Set<number>();

    // Segments first — collect which vertex indices are part of selected segments
    stadium.segments?.forEach((seg, i) => {
      const v0 = stadium.vertexes[seg.v0], v1 = stadium.vertexes[seg.v1];
      if (v0 && v1 && inside(v0.x, v0.y) && inside(v1.x, v1.y)) {
        result.push({ type: 'segment', index: i });
        selectedSegmentVertexes.add(seg.v0);
        selectedSegmentVertexes.add(seg.v1);
      }
    });

    // Vertices that are NOT already endpoint of a selected segment
    stadium.vertexes?.forEach((v, i) => {
      if (v.x != null && inside(v.x, v.y) && !selectedSegmentVertexes.has(i))
        result.push({ type: 'vertex', index: i });
    });

    stadium.discs?.forEach((d, i) => { if (d?.pos && inside(d.pos[0], d.pos[1])) result.push({ type: 'disc', index: i }); });
    stadium.goals?.forEach((g, i) => { if (g?.p0 && inside(g.p0[0], g.p0[1])) result.push({ type: 'goal', index: i }); });
    return result;
  }, []);

  // ── Context menu helpers ──────────────────────────────────────
  const getSelectionWorldPos = () => {
    const { selection, stadium } = stateRef.current;
    if (selection.length === 0) return null;
    const sel = selection[0];
    if (sel.type === 'vertex') { const v = stadium.vertexes[sel.index]; return v ? { x: v.x, y: v.y } : null; }
    if (sel.type === 'disc') { const d = stadium.discs[sel.index]; return d?.pos ? { x: d.pos[0], y: d.pos[1] } : null; }
    if (sel.type === 'goal') { const g = stadium.goals[sel.index]; return g?.p0 ? { x: g.p0[0], y: g.p0[1] } : null; }
    return null;
  };

  const applyPositionEdit = (px: number, py: number) => {
    const { selection, stadium } = stateRef.current;
    if (selection.length === 0) return;
    const sel = selection[0];
    if (sel.type === 'vertex') dispatch({ type: 'UPDATE_VERTEX', index: sel.index, payload: { x: px, y: py } });
    if (sel.type === 'disc') dispatch({ type: 'UPDATE_DISC', index: sel.index, payload: { pos: [px, py] } });
    if (sel.type === 'goal') {
      const g = stadium.goals[sel.index];
      if (g?.p0 && g?.p1) {
        const dx = px - g.p0[0], dy = py - g.p0[1];
        dispatch({ type: 'UPDATE_GOAL', index: sel.index, payload: { p0: [px, py], p1: [g.p1[0] + dx, g.p1[1] + dy] } });
      }
    }
  };

  // ── Mouse Events ─────────────────────────────────────────────
  const onMouseDown = (e: React.MouseEvent) => {
    if (contextMenu) { setContextMenu(null); setPosEdit(null); return; }

    const { tool, stadium } = stateRef.current;
    const { x, y } = screenToWorld(e.clientX, e.clientY);
    const { sx, sy } = screenCoordsRaw(e.clientX, e.clientY);

    if (e.button === 1 || tool === 'pan') {
      isPanningRef.current = true;
      panStartRef.current = { x: e.clientX, y: e.clientY, camX: stateRef.current.camX, camY: stateRef.current.camY };
      e.preventDefault();
      return;
    }

    if (tool === 'select') {
      const hit = getHoveredObject(x, y);
      if (hit) {
        if (!e.shiftKey) dispatch({ type: 'SET_SELECTION', payload: [hit] });
        else dispatch({ type: 'ADD_SELECTION', payload: hit });
        isDraggingRef.current = true;
        dragStartRef.current = { x, y };
      } else {
        if (!e.shiftKey) dispatch({ type: 'SET_SELECTION', payload: [] });
        rubberBandRef.current = { sx, sy, ex: sx, ey: sy };
        rubberBandActiveRef.current = true;
      }
    }

    if (tool === 'vertex') {
      // Create a minimal vertex — just x/y, no extra physics props
      const newV = { x: Math.round(x), y: Math.round(y) };
      dispatch({ type: 'UPDATE_STADIUM', payload: { vertexes: [...(stadium.vertexes ?? []), newV] } });
      dispatch({ type: 'SET_SELECTION', payload: [{ type: 'vertex', index: stadium.vertexes.length }] });
    }

    if (tool === 'segment') {
      const hit = getHoveredObject(x, y);
      let vIndex = hit?.type === 'vertex' ? hit.index : -1;
      if (vIndex === -1) {
        dispatch({ type: 'UPDATE_STADIUM', payload: { vertexes: [...(stadium.vertexes ?? []), { x: Math.round(x), y: Math.round(y) }] } });
        vIndex = stadium.vertexes.length;
      }
      if (tempVertexRef.current === null) {
        tempVertexRef.current = vIndex;
      } else {
        if (tempVertexRef.current !== vIndex) {
          dispatch({ type: 'UPDATE_STADIUM', payload: { segments: [...(stadium.segments ?? []), { v0: tempVertexRef.current, v1: vIndex, curve: 0, vis: true, color: 'FFFFFF', bCoef: 1 }] } });
          dispatch({ type: 'SET_SELECTION', payload: [{ type: 'segment', index: stadium.segments.length }] });
        }
        tempVertexRef.current = null;
        dispatch({ type: 'SET_TOOL', payload: 'select' });
      }
    }

    if (tool === 'disc') {
      dispatch({ type: 'UPDATE_STADIUM', payload: { discs: [...(stadium.discs ?? []), { pos: [Math.round(x), Math.round(y)] as [number, number], radius: 10, color: 'FFAA00', bCoef: 0.457, invMass: 1, damping: 0.99 }] } });
      dispatch({ type: 'SET_SELECTION', payload: [{ type: 'disc', index: stadium.discs.length }] });
      dispatch({ type: 'SET_TOOL', payload: 'select' });
    }

    if (tool === 'goal') {
      dispatch({ type: 'UPDATE_STADIUM', payload: { goals: [...(stadium.goals ?? []), { p0: [Math.round(x), Math.round(y - 50)] as [number, number], p1: [Math.round(x), Math.round(y + 50)] as [number, number], team: 'red' as const }] } });
      dispatch({ type: 'SET_SELECTION', payload: [{ type: 'goal', index: stadium.goals.length }] });
      dispatch({ type: 'SET_TOOL', payload: 'select' });
    }
  };

  const onMouseMove = (e: React.MouseEvent) => {
    const { x, y } = screenToWorld(e.clientX, e.clientY);
    const { sx, sy } = screenCoordsRaw(e.clientX, e.clientY);
    dispatch({ type: 'SET_MOUSE', payload: { x, y } });

    if (isPanningRef.current && panStartRef.current) {
      const dx = e.clientX - panStartRef.current.x, dy = e.clientY - panStartRef.current.y;
      const { zoom } = stateRef.current;
      dispatch({ type: 'SET_CAMERA', payload: { zoom, camX: panStartRef.current.camX - dx / zoom, camY: panStartRef.current.camY - dy / zoom } });
      return;
    }

    if (rubberBandActiveRef.current && rubberBandRef.current) {
      rubberBandRef.current = { ...rubberBandRef.current, ex: sx, ey: sy };
      return;
    }

    if (isDraggingRef.current && dragStartRef.current) {
      const dx = x - dragStartRef.current.x, dy = y - dragStartRef.current.y;
      if (dx === 0 && dy === 0) return;
      const { selection, stadium } = stateRef.current;
      const updates: Record<string, any> = {};
      selection.forEach(sel => {
        if (sel.type === 'vertex') {
          if (!updates.vertexes) updates.vertexes = [...(stadium.vertexes ?? [])];
          const v = updates.vertexes[sel.index];
          if (v) updates.vertexes[sel.index] = { ...v, x: Math.round(v.x + dx), y: Math.round(v.y + dy) };
        }
        if (sel.type === 'disc') {
          if (!updates.discs) updates.discs = [...(stadium.discs ?? [])];
          const d = updates.discs[sel.index];
          if (d?.pos) updates.discs[sel.index] = { ...d, pos: [Math.round(d.pos[0] + dx), Math.round(d.pos[1] + dy)] };
        }
        if (sel.type === 'goal') {
          if (!updates.goals) updates.goals = [...(stadium.goals ?? [])];
          const g = updates.goals[sel.index];
          if (g?.p0 && g?.p1) updates.goals[sel.index] = { ...g, p0: [Math.round(g.p0[0] + dx), Math.round(g.p0[1] + dy)] as [number, number], p1: [Math.round(g.p1[0] + dx), Math.round(g.p1[1] + dy)] as [number, number] };
        }
        if (sel.type === 'segment') {
          // Moving a segment moves both its vertices
          if (!updates.vertexes) updates.vertexes = [...(stadium.vertexes ?? [])];
          const seg = stadium.segments[sel.index];
          if (seg) {
            const v0 = updates.vertexes[seg.v0];
            if (v0) updates.vertexes[seg.v0] = { ...v0, x: Math.round(v0.x + dx), y: Math.round(v0.y + dy) };
            const v1 = updates.vertexes[seg.v1];
            if (v1) updates.vertexes[seg.v1] = { ...v1, x: Math.round(v1.x + dx), y: Math.round(v1.y + dy) };
          }
        }
      });
      if (Object.keys(updates).length > 0) dispatch({ type: 'UPDATE_STADIUM', payload: updates, pushHistory: false });
      dragStartRef.current = { x, y };
    }
  };

  const onMouseUp = () => {
    if (rubberBandActiveRef.current && rubberBandRef.current) {
      const rb = rubberBandRef.current;
      const canvas = canvasRef.current!;
      const { zoom, camX, camY, showRulers } = stateRef.current;
      const RULER = showRulers ? 20 : 0;
      const s2w = (sx: number, sy: number) => ({
        x: (sx - RULER - (canvas.width - RULER) / 2) / zoom + camX,
        y: (sy - RULER - (canvas.height - RULER) / 2) / zoom + camY,
      });
      const wA = s2w(rb.sx, rb.sy), wB = s2w(rb.ex, rb.ey);
      const inRect = getObjectsInRect(wA.x, wA.y, wB.x, wB.y);
      if (inRect.length > 0) dispatch({ type: 'SET_SELECTION', payload: inRect as any });
      rubberBandActiveRef.current = false;
      rubberBandRef.current = null;
      return;
    }
    if (isDraggingRef.current) dispatch({ type: 'UPDATE_STADIUM', payload: {} });
    isDraggingRef.current = false;
    dragStartRef.current = null;
    isPanningRef.current = false;
    panStartRef.current = null;
  };

  const onWheel = (e: React.WheelEvent) => {
    const delta = e.deltaY > 0 ? 0.9 : 1.1;
    const newZoom = Math.max(0.1, Math.min(20, stateRef.current.zoom * delta));
    dispatch({ type: 'SET_CAMERA', payload: { camX: stateRef.current.camX, camY: stateRef.current.camY, zoom: newZoom } });
  };

  const onContextMenu = (e: React.MouseEvent) => {
    e.preventDefault();
    const { x, y } = screenToWorld(e.clientX, e.clientY);
    const hit = getHoveredObject(x, y);
    if (hit) {
      const alreadySel = stateRef.current.selection.some(s => s.type === hit.type && s.index === hit.index);
      if (!alreadySel) dispatch({ type: 'SET_SELECTION', payload: [hit] });
    }
    const rect = canvasRef.current!.getBoundingClientRect();
    setContextMenu({ screenX: e.clientX - rect.left, screenY: e.clientY - rect.top, worldX: Math.round(x), worldY: Math.round(y), target: hit ?? null });
    setPosEdit(null);
  };

  const closeMenu = () => { setContextMenu(null); setPosEdit(null); };
  const menuCopy = () => { dispatch({ type: 'COPY_SELECTION' }); closeMenu(); };
  const menuDelete = () => { dispatch({ type: 'DELETE_SELECTION' }); closeMenu(); };
  const menuPaste = () => { dispatch({ type: 'PASTE_CLIPBOARD' }); closeMenu(); };
  const menuSetPos = () => { const pos = getSelectionWorldPos(); if (pos) setPosEdit({ x: String(Math.round(pos.x)), y: String(Math.round(pos.y)) }); };
  const menuApplyPos = () => { if (!posEdit) return; applyPositionEdit(Number(posEdit.x), Number(posEdit.y)); closeMenu(); };

  const hasClipboard = state.clipboard.length > 0;
  const selHasPos = state.selection.length > 0 && ['vertex', 'disc', 'goal'].includes(state.selection[0]?.type);

  const canvasW = canvasRef.current?.clientWidth ?? 800;
  const canvasH = canvasRef.current?.clientHeight ?? 600;
  const menuX = contextMenu ? Math.min(contextMenu.screenX, canvasW - 210) : 0;
  const menuY = contextMenu ? Math.min(contextMenu.screenY, canvasH - 230) : 0;

  return (
    <div className="flex-1 relative overflow-hidden select-none"
      style={{ background: 'var(--canvas-bg)', cursor: state.tool === 'pan' ? 'grab' : 'crosshair' }}>
      <canvas
        ref={canvasRef}
        className="block w-full h-full"
        onMouseDown={onMouseDown}
        onMouseMove={onMouseMove}
        onMouseUp={onMouseUp}
        onMouseLeave={onMouseUp}
        onWheel={onWheel}
        onContextMenu={onContextMenu}
      />

      {/* ── Right-click Context Menu ───────────────────────── */}
      {contextMenu && (
        <div
          className="absolute z-50 min-w-[200px] rounded-xl shadow-2xl border border-border overflow-hidden"
          style={{ left: menuX, top: menuY, background: 'var(--panel)' }}
          onMouseDown={e => e.stopPropagation()}
        >
          <div className="px-3 py-2 border-b border-border bg-background/60">
            <p className="text-[10px] font-bold text-muted-foreground uppercase tracking-wider">
              {contextMenu.target
                ? `${contextMenu.target.type} #${contextMenu.target.index}`
                : `(${contextMenu.worldX}, ${contextMenu.worldY})`}
            </p>
          </div>
          <div className="p-1">
            {contextMenu.target && (
              <>
                <button onClick={menuCopy} className="w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-sm hover:bg-primary/10 hover:text-primary text-foreground transition-colors text-left">
                  <Copy className="w-3.5 h-3.5" /> Copy
                  <span className="ml-auto text-[10px] text-muted-foreground">Ctrl+C</span>
                </button>
                <button onClick={menuDelete} className="w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-sm hover:bg-red-500/10 hover:text-red-400 text-foreground transition-colors text-left">
                  <Trash2 className="w-3.5 h-3.5" /> Delete
                  <span className="ml-auto text-[10px] text-muted-foreground">Del</span>
                </button>
                {selHasPos && !posEdit && (
                  <button onClick={menuSetPos} className="w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-sm hover:bg-muted text-foreground transition-colors text-left">
                    <MapPin className="w-3.5 h-3.5" /> Set Position…
                  </button>
                )}
                {posEdit && (
                  <div className="px-3 py-2 border-t border-border mt-1">
                    <p className="text-[10px] text-muted-foreground mb-2 font-medium uppercase tracking-wide">Set Position</p>
                    <div className="grid grid-cols-2 gap-1.5 mb-2">
                      <div>
                        <label className="text-[10px] text-muted-foreground">X</label>
                        <input type="number" value={posEdit.x} autoFocus
                          onChange={e => setPosEdit(p => p ? { ...p, x: e.target.value } : p)}
                          onKeyDown={e => { if (e.key === 'Enter') menuApplyPos(); if (e.key === 'Escape') closeMenu(); }}
                          className="w-full bg-background border border-border px-2 py-1 rounded text-xs focus:outline-none focus:border-primary" />
                      </div>
                      <div>
                        <label className="text-[10px] text-muted-foreground">Y</label>
                        <input type="number" value={posEdit.y}
                          onChange={e => setPosEdit(p => p ? { ...p, y: e.target.value } : p)}
                          onKeyDown={e => { if (e.key === 'Enter') menuApplyPos(); if (e.key === 'Escape') closeMenu(); }}
                          className="w-full bg-background border border-border px-2 py-1 rounded text-xs focus:outline-none focus:border-primary" />
                      </div>
                    </div>
                    <button onClick={menuApplyPos} className="w-full py-1.5 bg-primary text-primary-foreground rounded text-xs font-semibold hover:bg-primary/90 transition-colors">
                      Apply
                    </button>
                  </div>
                )}
                <div className="border-t border-border my-1" />
              </>
            )}
            <button onClick={menuPaste} disabled={!hasClipboard}
              className="w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-sm hover:bg-primary/10 hover:text-primary text-foreground transition-colors text-left disabled:opacity-30 disabled:cursor-not-allowed">
              <Clipboard className="w-3.5 h-3.5" /> Paste
              <span className="ml-auto text-[10px] text-muted-foreground">Ctrl+V</span>
            </button>
            {!contextMenu.target && (
              <button onClick={() => {
                const { stadium } = stateRef.current;
                const all: Selection[] = [];
                stadium.vertexes?.forEach((_, i) => all.push({ type: 'vertex', index: i }));
                stadium.discs?.forEach((_, i) => all.push({ type: 'disc', index: i }));
                stadium.goals?.forEach((_, i) => all.push({ type: 'goal', index: i }));
                dispatch({ type: 'SET_SELECTION', payload: all });
                closeMenu();
              }} className="w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-sm hover:bg-muted text-foreground transition-colors text-left">
                <MapPin className="w-3.5 h-3.5" /> Select All
              </button>
            )}
          </div>
        </div>
      )}

      {/* ── Coordinate overlay ──────────────────────────────── */}
      <div className="absolute bottom-3 left-3 right-3 flex justify-between pointer-events-none">
        <div className="bg-panel/90 backdrop-blur border border-border text-xs font-mono px-3 py-1.5 rounded-lg shadow text-foreground flex gap-4">
          <span>X: {Math.round(state.mouseX)}</span>
          <span>Y: {Math.round(state.mouseY)}</span>
        </div>
        <div className="bg-panel/90 backdrop-blur border border-border text-xs font-mono px-3 py-1.5 rounded-lg shadow text-foreground">
          {Math.round(state.zoom * 100)}%
        </div>
      </div>
    </div>
  );
}

      {children}
    </div>
  );
}

}

/** cMask displayed as comma-separated numbers (HBS allows number or array) */
function CMaskField({ value, onChange }: { value: number[] | number | undefined; onChange: (v: number[] | number) => void }) {
  const display = Array.isArray(value) ? value.join(', ') : value != null ? String(value) : '';
  return (
    <Field label="cMask">
      <input
        type="text"
        value={display}
        placeholder="e.g. 1, 2, 4"
        onChange={e => {
          const raw = e.target.value.trim();
          if (!raw) { onChange(0); return; }
          const parts = raw.split(',').map(s => Number(s.trim())).filter(n => !isNaN(n));
          onChange(parts.length === 1 ? parts[0] : parts);
        }}
        className="w-full bg-background border border-border px-2 py-1.5 rounded-lg text-sm focus:outline-none focus:border-primary font-mono transition-colors"
      />
    </Field>
  );
}

function TraitSelect({ value, onChange }: { value: string | undefined; onChange: (v: string) => void }) {
  return (
    <Field label="trait">
      <select
        value={value ?? ''}
        onChange={e => onChange(e.target.value)}
        className="w-full bg-background border border-border px-2 py-1.5 rounded-lg text-sm focus:outline-none focus:border-primary transition-colors"
      >
        {TRAITS.map(t => <option key={t} value={t}>{t || '(none)'}</option>)}
      </select>
    </Field>
  );
}

function ColorField({ value, onChange }: { value: string | undefined; onChange: (v: string) => void }) {
  const hex = '#' + (value || '000000');
  return (
    <Field label="color">
      <div className="flex items-center gap-2">
        <input
          type="color"
          value={hex}
          onChange={e => onChange(e.target.value.replace('#', '').toUpperCase())}
          className="w-10 h-9 rounded border-none cursor-pointer flex-shrink-0"
        />
        <TextInput value={value ?? ''} onChange={v => onChange(v.toUpperCase())} placeholder="FFFFFF" />
      </div>
    </Field>
  );
}

export function PropertiesPanel() {
  const { state } = useEditor();
  const sel = state.selection[0];

  if (!sel) {
    return (
      <div className="w-72 bg-panel border-l h-full flex flex-col items-center justify-center p-6 text-center shadow-lg z-10">
        <div className="w-16 h-16 rounded-full bg-muted flex items-center justify-center mb-4">
          <Dot className="w-8 h-8 text-muted-foreground" />
        </div>
        <h3 className="text-foreground font-medium">No object selected</h3>
        <p className="text-xs text-muted-foreground mt-2">Click on an object in the canvas to edit its properties.</p>
        {state.selection.length === 0 && (
          <p className="text-[10px] text-muted-foreground/60 mt-4">Tip: Right-click for more options</p>
        )}
      </div>
    );
  }

  return (
    <div className="w-72 bg-panel border-l h-full flex flex-col z-10 shadow-lg">
      {/* Header */}
      <div className="px-4 py-3 border-b bg-background/50 backdrop-blur shrink-0">
        <div className="flex items-center justify-between">
          <div className="flex items-center gap-2 text-foreground">
            {TYPE_ICON[sel.type]}
            <span className="text-sm font-bold capitalize">{sel.type}</span>
          </div>
          <span className="text-[10px] bg-primary/20 text-primary px-2 py-0.5 rounded-full font-mono">#{sel.index}</span>
        </div>
        {state.selection.length > 1 && (
          <p className="text-[10px] text-muted-foreground mt-1">{state.selection.length} objects selected — editing first</p>
        )}
      </div>

      {/* Properties */}
      <div className="flex-1 overflow-y-auto p-4">
        {sel.type === 'vertex'  && <VertexProps  index={sel.index} />}
        {sel.type === 'segment' && <SegmentProps index={sel.index} />}
        {sel.type === 'disc'    && <DiscProps    index={sel.index} />}
        {sel.type === 'goal'    && <GoalProps    index={sel.index} />}
        {sel.type === 'plane'   && <PlaneProps   index={sel.index} />}
      </div>
    </div>
  );
}

import React, { useState, useEffect } from 'react';
import { useEditor } from '@/context/EditorContext';
import { Save, X } from 'lucide-react';

export function TextModeEditor() {
  const { state, dispatch } = useEditor();
  const [text, setText] = useState('');
  const [error, setError] = useState('');

  useEffect(() => {
    setText(JSON.stringify(state.stadium, null, 2));
  }, [state.stadium, state.textMode]);

  if (!state.textMode) return null;

  const handleSave = () => {
    try {
      const parsed = JSON.parse(text);
      dispatch({ type: 'SET_STADIUM', payload: parsed });
      dispatch({ type: 'SET_TEXT_MODE', payload: false });
      setError('');
    } catch (err: any) {
      setError(err.message);
    }
  };

  return (
    <div className="absolute inset-0 z-50 bg-background flex flex-col">
      <div className="h-14 border-b bg-panel flex items-center justify-between px-4">
        <div className="flex items-center gap-4">
          <h2 className="font-bold text-primary">JSON Code Editor</h2>
          {error && <span className="text-red-500 text-sm font-medium">{error}</span>}
        </div>
        <div className="flex items-center gap-2">
          <button 
            onClick={handleSave}
            className="flex items-center gap-2 bg-primary text-primary-foreground px-4 py-2 rounded-lg font-medium hover:bg-primary/90 transition-colors"
          >
            <Save className="w-4 h-4" /> Save & Exit
          </button>
          <button 
            onClick={() => dispatch({ type: 'SET_TEXT_MODE', payload: false })}
            className="flex items-center gap-2 bg-muted text-foreground px-4 py-2 rounded-lg font-medium hover:bg-muted/80 transition-colors"
          >
            <X className="w-4 h-4" /> Cancel
          </button>
        </div>
      </div>
      <textarea
        className="flex-1 w-full p-4 bg-[#0d0d14] text-[#a5b4fc] font-mono text-sm outline-none resize-none"
        value={text}
        spellCheck={false}
        onChange={(e) => setText(e.target.value)}
      />
    </div>
  );
}
